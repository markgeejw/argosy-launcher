cmake_minimum_required(VERSION 3.22.1)
project(argosy-sigil-tests C)

get_filename_component(SIGIL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../sigil" ABSOLUTE)
set(SIGIL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SIGIL_BUILD_CLI OFF CACHE BOOL "" FORCE)
add_subdirectory("${SIGIL_ROOT}" sigil-build)
include("${CMAKE_CURRENT_SOURCE_DIR}/ApplyPatch.cmake")
argosy_patch_sigil()

enable_testing()
add_executable(unit_3ds_extdata tests/unit_3ds_extdata.c)
target_include_directories(unit_3ds_extdata PRIVATE "${SIGIL_ROOT}/src" "${SIGIL_ROOT}/tests")
target_link_libraries(unit_3ds_extdata PRIVATE sigil)
add_test(NAME unit_3ds_extdata COMMAND unit_3ds_extdata)

foreach(name unit_3ds unit_save_unit unit_zip unit_layouts unit_sync_wii integration_sync_profiles integration_sync_ps2)
    add_executable(${name} "${SIGIL_ROOT}/tests/${name}.c")
    target_include_directories(${name} PRIVATE "${SIGIL_ROOT}/src" "${SIGIL_ROOT}/tests")
    target_compile_definitions(${name} PRIVATE SIGIL_SAVE_FIXTURES="${SIGIL_ROOT}/tests/fixtures/saves")
    target_link_libraries(${name} PRIVATE sigil)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
target_link_libraries(unit_zip PRIVATE sigil_zlib)
